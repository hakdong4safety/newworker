Firebase 설정 — 순서대로 3단계
════════════════════════════════════════════════════════════

【1단계】 이메일 로그인 켜기
  Firebase 콘솔 → Authentication → 시작하기
  → Sign-in method → 이메일/비밀번호 → 사용 설정 → 저장
  → 설정(Settings) 탭 → 승인된 도메인 → 본인의 xxx.github.io 추가

【2단계】 아래 규칙 붙여넣기
  Firestore Database → 규칙(Rules) → 전체 지우고 붙여넣기 → [게시]
  ※ 게시 버튼을 누르지 않으면 반영되지 않습니다.

【3단계】 원청 계정 만들기  ★ 앱에서 합니다
  배포된 주소 접속 → 우측 위 [관리자]
  → 맨 아래 "현장을 처음 여시나요 · 원청 관리자 계정 만들기"
  → 아이디 hdcipark01 · 비밀번호 · 담당자 성함 → 만들기

규칙 (2단계에 붙여넣을 내용)
────────────────────────────────────────────────────────────

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() { return request.auth != null; }
    function known()  { return signedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid)); }
    function me()     { return get(/databases/$(database)/documents/users/$(request.auth.uid)).data; }
    function ok()     { return known() && me().approved == true; }
    function isGC()   { return ok() && me().role == 'gc'; }
    function myCo()   { return ok() ? me().company : '__none__'; }

    // ── 근로자 제출 기록 (1명당 문서 1건) ──
    match /records/{id} {
      function okRecord() {
        return request.resource.data.keys().hasOnly(['json','updatedAt','company','dateKey'])
            && request.resource.data.json is string
            && request.resource.data.json.size() < 1000000
            && request.resource.data.company is string;
      }
      allow create: if okRecord();                     // 근로자는 로그인 없이 제출만
      allow read:   if isGC() || (ok() && resource.data.company == myCo());
      allow update: if okRecord()
                    && (isGC() || (ok() && resource.data.company == myCo()));
      allow delete: if isGC();
    }

    // ── 원청 자리 (한 번에 한 계정만) ──
    match /meta/bootstrap {
      allow read:   if true;
      allow create: if signedIn() && request.resource.data.uid == request.auth.uid;
      allow update, delete: if isGC();               // 원청이 자리를 비워 아이디를 새로 만들 수 있게
    }

    // ── 협력사 고유번호 (원청이 발급 · 원청만 읽음) ──
    match /cocodes/{co} {
      allow read, write: if isGC();
    }

    // ── 계정 ──
    match /users/{uid} {
      allow read: if isGC() || (signedIn() && request.auth.uid == uid);

      // 협력사: 승인 대기 상태로만, 그리고 회사 고유번호가 맞아야 가입 가능
      //        (번호는 서버가 대조하므로 앱을 뜯어봐도 알 수 없다)
      // 원청:  bootstrap 자리를 잡은 사람만 가능
      allow create: if signedIn() && request.auth.uid == uid && (
           (request.resource.data.role == 'sub'
            && request.resource.data.approved == false
            && request.resource.data.code is string
            && exists(/databases/$(database)/documents/cocodes/$(request.resource.data.company))
            && get(/databases/$(database)/documents/cocodes/$(request.resource.data.company)).data.code
               == request.resource.data.code)
        || (request.resource.data.role == 'gc'
            && exists(/databases/$(database)/documents/meta/bootstrap)
            && get(/databases/$(database)/documents/meta/bootstrap).data.uid == request.auth.uid)
      );
      allow update, delete: if isGC();
    }

    // ── 양식 설정 · 협력사 이름 목록 · 순번 카운터 ──
    // 근로자 화면도 읽어야 하므로 읽기는 열어 둔다. 개인정보와 고유번호는 들어가지 않는다.
    match /kv/{id} {
      allow read:  if true;
      allow write: if isGC() || id.matches('seq-.*') || id == '__diag';
    }
  }
}

그 다음에 할 일
────────────────────────────────────────────────────────────

▸ 협력사 등록 (원청)
  로그인 → [협력사 등록] 탭 → 협력사명 입력 → 추가
  → 고유번호(예: ABC-234)가 자동 발급됩니다
  → [안내문 복사]로 그 업체 담당자에게 전달

▸ 협력사 담당자 가입
  같은 주소 → [관리자] → "협력사 담당자 계정 만들기"
  → 아이디 · 비밀번호 · 회사명 · 고유번호 · 성함
  → 원청이 [계정 관리]에서 승인하면 자기 회사 근로자만 보입니다

▸ 기존 자료 정비 (한 번만)
  [설정] 탭 → [소속 정보 채우기]
  → 권한 구분 전에 저장된 기록이 협력사에게도 보이게 됩니다

▸ 원청 아이디를 바꾸고 싶을 때 (콘솔에서)
  Authentication → 사용자 → 기존 원청 계정 삭제
  Firestore → 데이터 → meta → bootstrap 문서 삭제
  → 앱에서 "원청 관리자 계정 만들기"로 새 아이디 생성
